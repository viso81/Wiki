# gum
命令行下交互式命令

## 命令
# 多选(无限制)
gum choose --no-limit 1 2 3 4 5 6
# 多需(限制2个项目)
gum choose --limit 2 1 2 3 4 5 6
```shell
# 显示一个高亮边框，选择“Yes/No”，用左右切换，回车确认
# 选Yes -> 退出码 0
# 选No或按Esc/Ctrl+C -> 退出码非0
if gum config "确定删除吗？"
  echo "已确定"
  rm <文件>
else
  echo "已取消"
end

# 输入和获取变量并打印
set title (gum input --placeholder "标题")
set os (gum choose "linux" "windows" --header "系统")
echo $title
echo $os
```
